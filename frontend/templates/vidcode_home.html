def two_sum(nums, target):
    seen = {}
    for i, n in enumerate(nums):
        diff = target - n
        if diff in seen:
            return [seen[diff], i]
        seen[n] = i
    return []

# ▶ Output: [0, 1]
print(two_sum([2,7,11,15], 9))
